} function cleanupNetwork() { // Putus koneksi secara agresif agar tidak jadi zombie if (conn) { conn.close(); conn = null; } if (peer) { peer.disconnect(); // Putus dari server setTimeout(() => { if(peer) { peer.destroy(); // Hancurkan semua WebRTC peer = null; } }, 100); } isOnlineMultiplayer = false; myNetworkRole = null; } function disconnectAndGoHome() { if(isOnlineMultiplayer && conn && conn.open) { conn.send({ type: 'player_left' }); } setTimeout(() => { cleanupNetwork(); goToMainMenu(); }, 200); } function hostRoom() { cleanupNetwork(); const rawCode = generateRoomCode(); const fullPeerId = 'NEONTIC-' + rawCode; mpStatus.textContent = "Menyiapkan ruangan..."; '1 (Biru)' : '2 (Pink)'}`; roleStatus.style.display = 'block'; showScreen(screenGame); resetBoard(false); } // --- LOGIKA GAME --- cells.forEach(cell => cell.addEventListener('click', function() { const cellIndex = parseInt(this.getAttribute('data-index')); if (board[cellIndex] !== "" || !isGameRunning) return; if (isOnlineMultiplayer && currentPlayer !== myNetworkRole) { return; // Bukan giliranmu } processMove(cellIndex, currentPlayer); if (isOnlineMultiplayer && conn && conn.open) { conn.send({ type: 'move', index: cellIndex, player: myNetworkRole }); } })); restartBtn.addEventListener('click', () => { resetBoard(true); }); function processMove(index, playerNum) { const cell = document.querySelector(`.cell[data-index="${index}"]`); board[index] = playerNum; if (playerNum === "X") { cell.innerHTML = customSymbol1 ? `` : "X"; cell.classList.add('x'); } else { cell.innerHTML = customSymbol2 ? `` : "O"; cell.classList.add('o'); } let currentMoves = (playerNum === "X") ? movesX : movesO; currentMoves.push(index); if (currentMoves.length > 3) { let oldestMoveIndex = currentMoves.shift(); board[oldestMoveIndex] = ""; let oldestCell = document.querySelector(`.cell[data-index="${oldestMoveIndex}"]`); oldestCell.innerHTML = ""; oldestCell.classList.remove('x', 'o', 'warning'); } updateWarnings(); checkWinner(); } function updateWarnings() { cells.forEach(c => c.classList.remove('warning')); if (movesX.length === 3) { let oldestX = document.querySelector(`.cell[data-index="${movesX[0]}"]`); if(oldestX) oldestX.classList.add('warning'); } if (movesO.length === 3) { let oldestO = document.querySelector(`.cell[data-index="${movesO[0]}"]`); if(oldestO) oldestO.classList.add('warning'); } } function changePlayer() { currentPlayer = (currentPlayer === "X") ? "O" : "X"; updateStatusText(); } function updateStatusText() { let symbolDisplay = ""; if (currentPlayer === "X") { symbolDisplay = customSymbol1 ? `` : "X"; } else { symbolDisplay = customSymbol2 ? `` : "O"; } let extraText = ""; if (isOnlineMultiplayer) { extraText = (currentPlayer === myNetworkRole) ? " - Giliranmu!" : " - Menunggu Lawan..."; } statusText.innerHTML = `Giliran: ${symbolDisplay}${extraText}`; statusText.className = (currentPlayer === "X") ? 'status-x' : 'status-o'; } function checkWinner() { let roundWon = false; for (let i = 0; i < winConditions.length; i++) { const [a, b, c] = winConditions[i]; if (board[a] && board[a] === board[b] && board[a] === board[c]) { roundWon = true; break; } } if (roundWon) { let winnerSymbol = ""; if (currentPlayer === "X") winnerSymbol = customSymbol1 ? `` : "X"; else winnerSymbol = customSymbol2 ? `` : "O"; let winMsg = isOnlineMultiplayer ? (currentPlayer === myNetworkRole ? `🎉 KAMU MENANG! 🎉` : `😭 LAWAN MENANG 😭`) : `🎉 ${winnerSymbol} MENANG! 🎉`; statusText.innerHTML = winMsg; statusText.classList.add('win-text'); boardElement.classList.add(`win-${currentPlayer.toLowerCase()}`); cells.forEach(c => c.classList.remove('warning')); isGameRunning = false; } else { changePlayer(); } } function resetBoard(broadcast = false) { if (broadcast && isOnlineMultiplayer && conn && conn.open) { conn.send({ type: 'restart' }); } currentPlayer = "X"; board = ["", "", "", "", "", "", "", "", ""]; movesX = []; movesO = []; boardElement.classList.remove('win-x', 'win-o'); statusText.classList.remove('win-text'); cells.forEach(cell => { cell.innerHTML = ""; cell.classList.remove('x', 'o', 'warning'); }); updateStatusText(); // Memberikan sedikit jeda sebelum game bisa diklik lagi agar sync MP lancar setTimeout(() => { isGameRunning = true; }, 100); } // --- LOGIKA MENGGAMBAR DI CANVAS --- function setupCanvas(canvasId, color) { const canvas = document.getElementById(canvasId); const ctx = canvas.getContext('2d'); let isDrawing = false; ctx.lineWidth = 6; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.strokeStyle = color; function getCoordinates(e) { const rect = canvas.getBoundingClientRect(); const clientX = e.touches ? e.touches[0].clientX : e.clientX; const clientY = e.touches ? e.touches[0].clientY : e.clientY; return { x: (clientX - rect.left) * (canvas.width / rect.width), y: (clientY - rect.top) * (canvas.height / rect.height) }; } function startPosition(e) { e.preventDefault(); isDrawing = true; draw(e); } function endPosition(e) { if(e && e.cancelable) e.preventDefault(); isDrawing = false; ctx.beginPath(); } function draw(e) { if (!isDrawing) return; e.preventDefault(); const pos = getCoordinates(e); ctx.lineTo(pos.x, pos.y); ctx.stroke(); ctx.beginPath(); ctx.moveTo(pos.x, pos.y); } canvas.addEventListener('mousedown', startPosition); canvas.addEventListener('mouseup', endPosition); canvas.addEventListener('mousemove', draw); canvas.addEventListener('mouseleave', endPosition); canvas.addEventListener('touchstart', startPosition, { passive: false }); canvas.addEventListener('touchend', endPosition, { passive: false }); canvas.addEventListener('touchmove', draw, { passive: false }); } setupCanvas('canvas1', '#00f3ff'); setupCanvas('canvas2', '#ff00ff'); function clearCanvas(playerNum) { const canvas = document.getElementById(`canvas${playerNum}`); const ctx = canvas.getContext('2d'); ctx.clearRect(0, 0, canvas.width, canvas.height); } function isCanvasEmpty(canvas) { const ctx = canvas.getContext('2d'); return !new Uint32Array(ctx.getImageData(0, 0, canvas.width, canvas.height).data.buffer).some(color => color !== 0); }
